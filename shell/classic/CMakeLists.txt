cmake_minimum_required(VERSION 3.21)
project(RightConvertClassic LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "RightConvert classic shell integration requires Windows")
endif()
add_library(RightConvertClassic SHARED RightConvertClassic.cpp RightConvertClassic.def)
target_compile_features(RightConvertClassic PRIVATE cxx_std_17)
target_compile_definitions(RightConvertClassic PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(RightConvertClassic PRIVATE ole32 shell32)
set_target_properties(RightConvertClassic PROPERTIES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
)
if(MSVC)
    target_compile_options(RightConvertClassic PRIVATE /W4 /EHsc /permissive-)
    target_link_options(RightConvertClassic PRIVATE /DYNAMICBASE /NXCOMPAT)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(RightConvertClassicSmoke ClassicSmoke.cpp)
    target_compile_features(RightConvertClassicSmoke PRIVATE cxx_std_17)
    target_compile_definitions(RightConvertClassicSmoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(RightConvertClassicSmoke PRIVATE shell32 ole32)
    if(MSVC)
        target_compile_options(RightConvertClassicSmoke PRIVATE /W4 /EHsc /permissive-)
    endif()
    add_test(NAME ClassicMultiSelect
        COMMAND RightConvertClassicSmoke $<TARGET_FILE:RightConvertClassic>)
endif()
